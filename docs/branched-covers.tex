\smalldoc{Branched covers of Riemann surfaces}{BrCov}
 

A non-constant holomorphic map $f\colon X\to Y$ of compact Riemann surfaces is a
covering map away from finitely many branch points (compare \docref{Cov}).

\begin{theorem}[Riemann--Hurwitz]
If $f$ has degree $n$ and ramification index $e_p$ at $p\in X$, then
\begin{equation}\label{eq:RH}
  2g_X - 2 = n\,(2g_Y - 2) + \sum_{p\in X} (e_p - 1).
\end{equation}
\end{theorem}
